#_|VOP优化|添加trigger,修改effect
country_event = {
	id = shroud.1505
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_technology = tech_aura_resonation
		any_owned_starbase = {
			has_starbase_aura_emitter = yes
			solar_system = {
				is_psionic_aura_suppressed = no
			}
		}
	}

	immediate = {
		every_owned_starbase = {
			limit = {
				has_starbase_aura_emitter = yes
				solar_system = {
					is_psionic_aura_suppressed = no
				}
			}
			solar_system = {
				natural_spawn_psionic_aura = { OWNER = root }
			}
		}
		every_owned_nonprimary_starbase = {
			limit = {
				has_starbase_aura_emitter = yes
				solar_system = {
					is_psionic_aura_suppressed = no
				}
			}
			solar_system = {
				natural_spawn_psionic_aura = { OWNER = root }
			}
		}
	}
}

#_|VOP优化|on_building_ruined 前置建筑类型过滤上提至 trigger
carrier_event = {
	id = shroud.9000
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		OR = {
			last_building_changed = building_psionic_suppressor
			last_building_changed = building_ancient_ward_2
		}
	}

	immediate = {
		if = {
			limit = { last_building_changed = building_psionic_suppressor }
			if = {
				limit = { has_active_building = building_psionic_suppressor }
				add_eotc_aura_psionic_suppressor = yes
			}
			else = {
				remove_eotc_aura_psionic_suppressor = yes
			}
		}
		else_if = {
			limit = { last_building_changed = building_ancient_ward_2 }
			if = {
				limit = { has_active_building = building_ancient_ward_2 }
				add_eotc_aura_grand_ward = yes
			}
			else = {
				remove_eotc_aura_grand_ward = yes
			}
		}
	}
}

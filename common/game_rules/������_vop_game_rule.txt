# Root = Pop Group
# From = Country
# This = Species
# NOTE: species_has_happiness is also checked for this
#_|VOP优化|三档思潮预设：默认仅玩家开启（兼容4.6意识形态/绝对思潮且省去全银河AI思潮分裂），可切换全开或全关
pop_group_has_random_ethics = {
	if = {
		limit = { has_global_flag = VOP_flg_glbl_disable_ethic_all }
		always = no
	}
	else_if = {
		limit = { has_global_flag = VOP_flg_glbl_enbale_ethic }
		exists = from
		NOT = { has_citizenship_type = { country = from type = citizenship_organic_trophy } }
	}
	else = {
		exists = from
		from? = { is_ai = no }
		NOT = { has_citizenship_type = { country = from type = citizenship_organic_trophy } }
	}
}

#Root = country
#This = pop_group
#_|VOP优化|联动三档思潮预设：默认仅玩家开启时短路AI人口组派系归属计算，全关时全银河短路
pop_group_can_join_factions = {
	if = {
		limit = { has_global_flag = VOP_flg_glbl_disable_ethic_all }
		always = no
	}
	else_if = {
		limit = {
			NOT = { has_global_flag = VOP_flg_glbl_enbale_ethic }
			owner? = { is_ai = yes }
		}
		always = no
	}
	else_if = {
		limit = {
			owner? = {
				is_democratic_authority = yes
				has_country_flag = cyber_collectivist
			}
		}
		OR = {
			pop_group_has_happiness = yes
			has_trait = trait_cybernetic
		}
	}
	else = {
		pop_group_has_happiness = yes
	}
}

				has_active_tradition = tr_cybernetics_integrated_anatomy
				has_active_tradition = tr_cybernetics_assimilator_integrated_anatomy
				has_active_tradition = tr_purity_genotype_regeneration
				has_active_tradition = tr_purity_genotype_regeneration_wilderness
				has_active_tradition = tr_cloning_somatic_synthesization
				has_active_tradition = tr_cloning_somatic_synthesization_clone_army
				has_active_tradition = tr_mutation_unnatural_selection
			}
		}
		AND = {
			has_toxoids = yes
			has_origin = origin_overtuned
		}

		has_technology = tech_epigenetic_triggers
	}
